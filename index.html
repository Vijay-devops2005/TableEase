MySQL DDL Schema (College Submission)

CREATE DATABASE tableease_db; USE tableease_db; -- 1. Customers Table CREATE TABLE customers ( id BIGINT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(100) NOT NULL, email VARCHAR(120) NOT NULL UNIQUE, phone VARCHAR(25) NOT NULL, notes TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); -- 2. Hotels & Restaurants Table CREATE TABLE restaurants ( id BIGINT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(120) NOT NULL, location VARCHAR(200) NOT NULL, cuisine VARCHAR(100) NOT NULL, rating DECIMAL(2,1) DEFAULT 4.5, contact VARCHAR(30), opening_hours VARCHAR(60), hero_image VARCHAR(255), google_maps_url VARCHAR(255) ); -- 3. Restaurant Tables CREATE TABLE restaurant_tables ( id BIGINT AUTO_INCREMENT PRIMARY KEY, restaurant_id BIGINT NOT NULL, table_number VARCHAR(20) NOT NULL, capacity INT NOT NULL, status ENUM('Available','Reserved','Occupied','Maintenance') DEFAULT 'Available', floor_area VARCHAR(50), FOREIGN KEY (restaurant_id) REFERENCES restaurants(id) ON DELETE CASCADE ); -- 4. Reservations Table CREATE TABLE reservations ( id BIGINT AUTO_INCREMENT PRIMARY KEY, customer_id BIGINT NOT NULL, restaurant_id BIGINT NOT NULL, table_id BIGINT NOT NULL, reservation_date DATE NOT NULL, reservation_time TIME NOT NULL, guest_count INT NOT NULL, status ENUM('Reserved','Occupied','Completed','Cancelled') DEFAULT 'Reserved', special_requests TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (customer_id) REFERENCES customers(id), FOREIGN KEY (restaurant_id) REFERENCES restaurants(id), FOREIGN KEY (table_id) REFERENCES restaurant_tables(id) ); -- 5. Menu Items Table CREATE TABLE menu_items ( id BIGINT AUTO_INCREMENT PRIMARY KEY, restaurant_id BIGINT, name VARCHAR(120) NOT NULL, category VARCHAR(50) NOT NULL, price DECIMAL(8,2) NOT NULL, is_veg BOOLEAN DEFAULT TRUE, rating DECIMAL(2,1) DEFAULT 4.8, short_desc VARCHAR(255), image_url VARCHAR(255) );

Spring Boot JdbcTemplate DAO Example

@Repository public class ReservationDaoImpl implements ReservationDao { @Autowired private JdbcTemplate jdbcTemplate; @Override public List findTodayReservations(LocalDate date) { String sql = "SELECT * FROM reservations WHERE reservation_date = ?"; return jdbcTemplate.query(sql, new ReservationRowMapper(), date); } @Override public int updateStatus(Long id, String status) { String sql = "UPDATE reservations SET status = ? WHERE id = ?"; return jdbcTemplate.update(sql, status, id); } @Override public int createReservation(Reservation res) { String sql = "INSERT INTO reservations " + "(customer_id, restaurant_id, table_id, " + "reservation_date, reservation_time, guest_count, status) " + "VALUES (?, ?, ?, ?, ?, ?, ?)"; return jdbcTemplate.update(sql, res.getCustomerId(), res.getRestaurantId(), res.getTableId(), res.getReservationDate(), res.getReservationTime(), res.getGuestCount(), res.getStatus()); } }

Viva Defense & Presentation Cheatsheet

Q: Why use JdbcTemplate instead of heavy JPA/Hibernate?
JdbcTemplate eliminates boilerplate JDBC code (connections, statements, exception handling) while retaining explicit, transparent SQL control and minimal runtime overhead for a college mini project.
Q: How does the system prevent table overbooking?
Before confirming a booking, the Service checks whether the selected table has any overlapping reservation for the same date/time window. Upon confirmation, the table status is updated to 'Reserved'.
Q: How are Google Maps integrated without paid API keys?
Direct Google Maps search URLs are constructed dynamically using verified Pune hotel addresses (`maps/search/?api=1&query=...`), enabling 1-click external navigation on all platforms without billing keys.
Q: How does the multi-layer architecture maintain Clean Architecture?
Presentation (HTML/CSS/JS) is decoupled from Business Logic (Service). The Controller handles HTTP JSON contracts, the Service enforces table validation rules, and DAO handles raw database persistence.
